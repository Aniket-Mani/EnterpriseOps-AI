def plan_warehouse_allocation(
    required_items: list[dict],
    inventory_result: dict,
) -> dict:
    """
    Build a deterministic, read-only warehouse allocation plan.

    Allocation priority:
    1. Preserve reorder levels wherever possible.
    2. Use stock from warehouses already below reorder level.
    3. Minimize new reorder-level violations.

    This function does not reserve, transfer, or modify inventory.
    """

    if not required_items:
        raise ValueError("Required items cannot be empty.")

    if not inventory_result.get("success"):
        raise ValueError("A successful inventory investigation is required.")

    inventory_by_sku = {
        product["sku"]: product
        for product in inventory_result.get("products", [])
    }

    allocations = []
    overall_sufficient = True
    warnings = []

    # Combine duplicate SKU requirements.
    requirements = {}

    for item in required_items:
        sku = item.get("sku")
        quantity = item.get("quantity")

        if not isinstance(sku, str) or not sku.strip():
            raise ValueError("Every item must have a valid SKU.")

        if (
            not isinstance(quantity, int)
            or isinstance(quantity, bool)
            or quantity <= 0
        ):
            raise ValueError(f"Invalid quantity for SKU {sku}.")

        sku = sku.strip()
        requirements[sku] = requirements.get(sku, 0) + quantity

    for sku, required_quantity in requirements.items():
        product = inventory_by_sku.get(sku)

        if not product or not product.get("inventory_found"):
            allocations.append({
                "sku": sku,
                "required_quantity": required_quantity,
                "allocated_quantity": 0,
                "shortfall": required_quantity,
                "warehouse_allocations": [],
                "sufficient": False,
            })
            overall_sufficient = False
            continue

        # Work on copies. Never mutate the inventory investigation.
        warehouses = []

        for warehouse in product.get("warehouses", []):
            available = warehouse["available_quantity"]
            reorder_level = warehouse["reorder_level"]

            if (
                not isinstance(available, int)
                or isinstance(available, bool)
                or not isinstance(reorder_level, int)
                or isinstance(reorder_level, bool)
                or available < 0
                or reorder_level < 0
            ):
                raise ValueError(
                    f"Invalid inventory data for SKU {sku}."
                )

            warehouses.append({
                "warehouse_code": warehouse["warehouse_code"],
                "warehouse_name": warehouse["warehouse_name"],
                "available_quantity": available,
                "reorder_level": reorder_level,
                "remaining_quantity": available,
            })

        # Phase 1: use stock above reorder levels.
        safe_priority = sorted(
            warehouses,
            key=lambda warehouse: (
                -max(
                    0,
                    warehouse["available_quantity"]
                    - warehouse["reorder_level"],
                ),
                warehouse["warehouse_code"],
            ),
        )

        remaining_required = required_quantity
        warehouse_allocations = []

        def allocate_from(warehouse, quantity):
            """Allocate stock and consolidate warehouse entries."""
            existing = next(
                (
                    allocation
                    for allocation in warehouse_allocations
                    if allocation["warehouse_code"]
                    == warehouse["warehouse_code"]
                ),
                None,
            )

            if existing:
                existing["allocated_quantity"] += quantity
            else:
                warehouse_allocations.append({
                    "warehouse_code": warehouse["warehouse_code"],
                    "warehouse_name": warehouse["warehouse_name"],
                    "allocated_quantity": quantity,
                })

            warehouse["remaining_quantity"] -= quantity

        for warehouse in safe_priority:
            if remaining_required <= 0:
                break

            safe_quantity = max(
                0,
                warehouse["remaining_quantity"]
                - warehouse["reorder_level"],
            )

            quantity_to_allocate = min(
                safe_quantity,
                remaining_required,
            )

            if quantity_to_allocate > 0:
                allocate_from(warehouse, quantity_to_allocate)
                remaining_required -= quantity_to_allocate

        # Phase 2: use stock from warehouses already below
        # their reorder level before creating new violations.
        already_below_priority = sorted(
            [
                warehouse
                for warehouse in warehouses
                if warehouse["remaining_quantity"] > 0
                and warehouse["remaining_quantity"]
                < warehouse["reorder_level"]
            ],
            key=lambda warehouse: (
                warehouse["remaining_quantity"],
                warehouse["warehouse_code"],
            ),
        )

        for warehouse in already_below_priority:
            if remaining_required <= 0:
                break

            quantity_to_allocate = min(
                warehouse["remaining_quantity"],
                remaining_required,
            )

            if quantity_to_allocate > 0:
                allocate_from(warehouse, quantity_to_allocate)
                remaining_required -= quantity_to_allocate

        # Phase 3: if still necessary, consume stock that would
        # newly push a warehouse below its reorder level.
        remaining_priority = sorted(
            [
                warehouse
                for warehouse in warehouses
                if warehouse["remaining_quantity"] > 0
            ],
            key=lambda warehouse: (
                -warehouse["remaining_quantity"],
                warehouse["warehouse_code"],
            ),
        )

        for warehouse in remaining_priority:
            if remaining_required <= 0:
                break

            quantity_to_allocate = min(
                warehouse["remaining_quantity"],
                remaining_required,
            )

            if quantity_to_allocate > 0:
                allocate_from(warehouse, quantity_to_allocate)
                remaining_required -= quantity_to_allocate

        # Calculate final remaining stock and reorder impact.
        for allocation in warehouse_allocations:
            warehouse = next(
                item for item in warehouses
                if item["warehouse_code"] == allocation["warehouse_code"]
            )

            remaining_available = warehouse["remaining_quantity"]
            reorder_level = warehouse["reorder_level"]

            below_reorder = remaining_available < reorder_level

            allocation["remaining_available"] = remaining_available
            allocation["reorder_level"] = reorder_level
            allocation["below_reorder_level_after_allocation"] = below_reorder

            if below_reorder:
                warnings.append(
                    f"{sku} allocation leaves "
                    f"{warehouse['warehouse_code']} below its "
                    "reorder level."
                )

        allocated_quantity = required_quantity - remaining_required
        sufficient = remaining_required == 0

        if not sufficient:
            overall_sufficient = False

        allocations.append({
            "sku": sku,
            "product_name": product.get("product_name"),
            "required_quantity": required_quantity,
            "allocated_quantity": allocated_quantity,
            "shortfall": remaining_required,
            "warehouse_allocations": warehouse_allocations,
            "sufficient": sufficient,
        })

    return {
        "agent": "allocation",
        "success": True,
        "overall_sufficient": overall_sufficient,
        "items": allocations,
        "warnings": warnings,
        "read_only": True,
        "execution_performed": False,
    }

from sqlmodel import Session, select
from sqlalchemy.orm import selectinload

from app.db.models import Order
from app.agents.inventory_agent import investigate_inventory
from app.agents.allocation_agent import plan_warehouse_allocation
from app.agents.procurement_agent import investigate_procurement


def evaluate_fulfillment(
    session: Session,
    order_number: str,
    inventory_result: dict | None = None,
    procurement_result: dict | None = None,
) -> dict:
    """
    Build a read-only fulfillment recommendation from independently
    gathered order, inventory, and procurement information.
    """

    if not order_number or not order_number.strip():
        raise ValueError("Order number is required.")

    statement = (
        select(Order)
        .where(Order.order_number == order_number.strip())
        .options(selectinload(Order.items))
    )
    order = session.exec(statement).first()

    if order is None:
        return {
            "agent": "fulfillment",
            "success": False,
            "error": f"Order {order_number} was not found.",
        }

    if not order.items:
        return {
            "agent": "fulfillment",
            "success": False,
            "error": f"Order {order_number} has no items.",
        }

    required_items = [
        {"sku": item.product_sku, "quantity": item.quantity}
        for item in order.items
    ]

    if inventory_result is None:
        inventory_result = investigate_inventory(session, required_items)

    if not inventory_result.get("success"):
        return {
            "agent": "fulfillment",
            "success": False,
            "error": "Inventory investigation failed.",
            "inventory_investigation": inventory_result,
        }

    allocation_result = plan_warehouse_allocation(
        required_items,
        inventory_result,
    )

    allocation_by_sku = {
        item["sku"]: item for item in allocation_result["items"]
    }
    inventory_by_sku = {
        product["sku"]: product
        for product in inventory_result["products"]
    }

    fulfillment_items = []
    procurement_requirements = []

    for item in required_items:
        sku = item["sku"]
        required = item["quantity"]

        inventory = inventory_by_sku.get(sku, {})
        allocation = allocation_by_sku.get(sku, {})

        allocated = allocation.get("allocated_quantity", 0)
        shortfall = max(0, required - allocated)

        fulfillment_items.append({
            "sku": sku,
            "required_quantity": required,
            "available_inventory": inventory.get("total_available", 0),
            "allocated_quantity": allocated,
            "inventory_shortfall": shortfall,
            "inventory_sufficient": shortfall == 0,
            "warehouse_allocations": allocation.get(
                "warehouse_allocations", []
            ),
            "allocation_warnings": [
                warning
                for warning in allocation_result.get("warnings", [])
                if sku in warning
            ],
        })

        if shortfall > 0:
            procurement_requirements.append({
                "sku": sku,
                "quantity": shortfall,
            })

    if procurement_requirements and procurement_result is None:
        procurement_result = investigate_procurement(
            session,
            procurement_requirements,
        )

    if procurement_result is not None and not procurement_result.get("success"):
        return {
            "agent": "fulfillment",
            "success": False,
            "error": "Procurement investigation failed.",
            "inventory_investigation": inventory_result,
            "allocation_plan": allocation_result,
            "procurement_investigation": procurement_result,
        }

    procurement_by_sku = {
        product["sku"]: product
        for product in (procurement_result or {}).get("products", [])
    }

    for item in fulfillment_items:
        if item["inventory_sufficient"]:
            item["recommended_action"] = "fulfill_from_allocated_inventory"
            item["supplier_options"] = []
            continue

        procurement = procurement_by_sku.get(item["sku"], {})
        eligible_options = [
            option
            for option in procurement.get("supplier_options", [])
            if option.get("eligible")
        ]

        item["supplier_options"] = eligible_options
        item["recommended_action"] = (
            "procure_shortfall"
            if eligible_options
            else "requires_manual_review"
        )

    all_inventory_sufficient = all(
        item["inventory_sufficient"] for item in fulfillment_items
    )
    all_items_have_path = all(
        item["recommended_action"] != "requires_manual_review"
        for item in fulfillment_items
    )

    if all_inventory_sufficient:
        overall_status = "inventory_fulfillment_possible"
    elif all_items_have_path:
        overall_status = "procurement_required"
    else:
        overall_status = "manual_review_required"

    return {
        "agent": "fulfillment",
        "success": True,
        "order_number": order.order_number,
        "order_status": order.status.value,
        "overall_status": overall_status,
        "items": fulfillment_items,
        "inventory_investigation": inventory_result,
        "allocation_plan": allocation_result,
        "procurement_investigation": procurement_result,
        "read_only": True,
        "execution_performed": False,
    }

def timestamp():
    return datetime.now(timezone.utc).isoformat()


def get_order_number(state: EnterpriseOpsState) -> str:
    order_number = (
        state.get("order_number")
        or state.get("task_context", {}).get("order_number")
    )
    if not order_number:
        raise ValueError("An order number is required.")
    return order_number.strip()


def orchestrator_node(state: EnterpriseOpsState) -> dict:
    try:
        objective = state.get("objective") or (
            f"Investigate and evaluate fulfillment for order "
            f"{get_order_number(state)}."
        )

        plan = create_execution_plan(
            objective=objective,
            context=state.get("task_context", {}),
        )

        return {
            "execution_plan": plan,
            "completed_steps": [],
            "step_results": {},
            "task_status": "running",
            "messages": [{
                "sender": "orchestrator",
                "content": "Validated execution plan created.",
                "timestamp": timestamp(),
            }],
            "agent_results": [{
                "agent": "orchestrator",
                "success": True,
                "result": plan,
            }],
            "audit_events": [{
                "event": "execution_plan_created",
                "timestamp": timestamp(),
                "agents_required": plan["agents_required"],
            }],
        }
    except Exception as exc:
        return {
            "errors": [f"Orchestrator error: {exc}"],
            "task_status": "failed",
        }



# Each handler receives a live database session.
def run_agent(agent: str, state: EnterpriseOpsState, session: Session) -> dict:
    order_number = get_order_number(state)

    if agent == "order":
        return investigate_order(session, order_number)

    if agent == "inventory":
        order_result = state.get("order_result") or {}
        if not order_result.get("success"):
            raise ValueError("Successful order investigation is required.")

        required_items = [
            {"sku": item["sku"], "quantity": item["quantity"]}
            for item in order_result.get("items", [])
        ]
        return investigate_inventory(session, required_items)

    if agent == "procurement":
        inventory = state.get("inventory_result") or {}
        if not inventory.get("success"):
            raise ValueError("Successful inventory investigation is required.")

        requirements = [
            {"sku": product["sku"], "quantity": product["shortfall"]}
            for product in inventory.get("products", [])
            if product.get("shortfall", 0) > 0
        ]

        if not requirements:
            return {
                "agent": "procurement",
                "success": True,
                "not_required": True,
                "products": [],
                "read_only": True,
            }

        return investigate_procurement(session, requirements)

    if agent == "fulfillment":
        return evaluate_fulfillment(
            session,
            order_number,
            inventory_result=state.get("inventory_result"),
            procurement_result=state.get("procurement_result"),
        )

    if agent == "verification":
        fulfillment = state.get("fulfillment_result")
        if not fulfillment:
            raise ValueError("Fulfillment result is required.")
        return verify_fulfillment(session, fulfillment)

    raise ValueError(f"Unsupported agent in execution plan: {agent}")


def execute_next_step_node(state: EnterpriseOpsState) -> dict:
    try:
        plan = state.get("execution_plan") or {}
        steps = plan.get("execution_plan", [])
        completed = list(state.get("completed_steps", []))
        step_results = dict(state.get("step_results", {}))

        if len(completed) == len(steps):
            return {}

        next_step = next(
            (
                step for step in steps
                if step["step"] not in completed
                and all(
                    dependency in completed
                    for dependency in step.get("depends_on", [])
                )
            ),
            None,
        )

        if next_step is None:
            raise ValueError(
                "Execution plan cannot progress: unresolved dependencies."
            )

        agent = next_step["agent"]

        with Session(engine) as session:
            result = run_agent(agent, state, session)

        if not result.get("success"):
            raise RuntimeError(
                result.get("error", f"{agent} agent reported failure.")
            )

        result_key = f"{agent}_result"
        step_results[str(next_step["step"])] = result
        completed.append(next_step["step"])

        return {
            result_key: result,
            "step_results": step_results,
            "completed_steps": completed,
            "current_agent": agent,
            "task_status": "running",
            "agent_results": [{
                "agent": agent,
                "success": True,
                "step": next_step["step"],
                "result": result,
            }],
            "messages": [{
                "sender": agent,
                "content": f"Execution step {next_step['step']} completed.",
                "timestamp": timestamp(),
            }],
            "audit_events": [{
                "event": "agent_step_completed",
                "agent": agent,
                "step": next_step["step"],
                "timestamp": timestamp(),
            }],
        }

    except Exception as exc:
        return {
            "errors": [f"Agent execution error: {exc}"],
            "task_status": "failed",
            "audit_events": [{
                "event": "agent_step_failed",
                "error": str(exc),
                "timestamp": timestamp(),
            }],
        }


def route_after_step(state: EnterpriseOpsState) -> str:
    if state.get("task_status") == "failed":
        return "report"

    plan = state.get("execution_plan") or {}
    steps = plan.get("execution_plan", [])
    completed = state.get("completed_steps", [])

    if len(completed) < len(steps):
        return "execute"

    if not state.get("verification_result", {}).get("verified"):
        return "report"

    if state.get("task_id") and plan.get("requires_human_approval"):
        return "approval"

    return "report"


def approval_node(state: EnterpriseOpsState) -> dict:
    try:
        plan = state.get("execution_plan") or {}
        verification = state.get("verification_result") or {}

        if not plan.get("requires_human_approval"):
            return {"requires_human_approval": False}

        if not verification.get("verified"):
            return {
                "task_status": "needs_review",
                "errors": [
                    "Approval was not created because verification failed."
                ],
            }

        task_id = state.get("task_id")
        if not task_id:
            raise ValueError("Task ID is required for persistent approval.")

        order_number = get_order_number(state)
        fulfillment = state.get("fulfillment_result") or {}
        action_type = "review_fulfillment_recommendation"

        with Session(engine) as session:
            existing = session.exec(
                sa_select(ApprovalRequest).where(
                    ApprovalRequest.task_id == task_id,
                    ApprovalRequest.action_type == action_type,
                    ApprovalRequest.status != ApprovalStatus.CANCELLED,
                )
            ).scalars().first()

            if existing:
                approval = existing
            else:
                approval = ApprovalRequest(
                    task_id=task_id,
                    agent_run_id=None,
                    action_type=action_type,
                    description=(
                        f"Review the verified fulfillment recommendation "
                        f"for order {order_number}. This does not authorize "
                        "operational execution."
                    ),
                    requested_action={
                        "order_number": order_number,
                        "action": action_type,
                        "fulfillment_evaluation": fulfillment,
                        "verification": verification,
                        "simulation": True,
                        "execution_enabled": False,
                    },
                    status=ApprovalStatus.PENDING,
                )
                session.add(approval)
                session.commit()
                session.refresh(approval)

            approval_data = {
                "approval_id": str(approval.id),
                "task_id": str(task_id),
                "action_type": approval.action_type,
                "status": approval.status.value,
                "execution_enabled": False,
            }

        return {

def report_node(state: EnterpriseOpsState) -> dict:
    errors = state.get("errors", [])
    verification = state.get("verification_result") or {}
    plan = state.get("execution_plan") or {}
    completed_steps = state.get("completed_steps", [])
    steps = plan.get("execution_plan", [])
    pending_approvals = state.get("pending_approvals", [])

    if errors:
        status = "failed"
    elif pending_approvals:
        status = "waiting_for_approval"
    elif (
        len(completed_steps) == len(steps)
        and verification.get("verified") is True
    ):
        status = "completed"
    else:
        status = "needs_review"

    report = {
        "objective": state.get("objective"),
        "order_number": state.get("order_number"),

        "execution_performed": False,
        "generated_at": timestamp(),
        "pending_approvals": pending_approvals,
        "requires_human_approval": bool(pending_approvals),
    }

    }


def build_graph(checkpointer=None):
    workflow = StateGraph(EnterpriseOpsState)

    workflow.add_node("orchestrator", orchestrator_node)
    workflow.add_node("execute_step", execute_next_step_node)
    workflow.add_node("approval", approval_node)
    workflow.add_node("report", report_node)

    workflow.add_edge(START, "orchestrator")

    workflow.add_conditional_edges(
        "orchestrator",
        lambda state: (
            "execute_step"
            if state.get("task_status") != "failed"
            else "report"
        ),
        {
            "execute_step": "execute_step",
            "report": "report",
        },
    )

    workflow.add_conditional_edges(
        "execute_step",
        route_after_step,
        {
            "execute": "execute_step",
            "approval": "approval",
            "report": "report",
        },
    )

    workflow.add_edge("approval", "report")
    workflow.add_edge("report", END)

    return workflow.compile(checkpointer=checkpointer)

from collections import defaultdict

from sqlmodel import Session, select

from app.db.models import Inventory, Warehouse


def investigate_inventory(
    session: Session,
    required_items: list[dict],
) -> dict:
    """
    Read-only inventory investigation.

    Checks whether requested quantities are available across
    all active warehouses. Does not reserve, transfer, or
    modify inventory.
    """

    if not required_items:
        raise ValueError("Required items cannot be empty.")

    # 1. Validate input and combine duplicate SKUs.
    requirements = {}

    for item in required_items:
        sku = item.get("sku")
        quantity = item.get("quantity")

        if not isinstance(sku, str) or not sku.strip():
            raise ValueError("Every item must have a valid SKU.")

        if (
            not isinstance(quantity, int)
            or isinstance(quantity, bool)
            or quantity <= 0
        ):
            raise ValueError(f"Invalid quantity for SKU {sku}.")

        sku = sku.strip()
        requirements[sku] = requirements.get(sku, 0) + quantity

    # 2. Fetch all requested SKUs in ONE database query.
    statement = (
        select(Inventory, Warehouse)
        .join(
            Warehouse,
            Inventory.warehouse_id == Warehouse.id,
        )
        .where(
            Inventory.product_sku.in_(list(requirements.keys())),
            Warehouse.is_active.is_(True),
        )
        .order_by(
            Inventory.product_sku,
            Warehouse.warehouse_code,
        )
    )

    records = session.exec(statement).all()

    # 3. Group inventory records by SKU.
    inventory_by_sku = defaultdict(list)

    for inventory, warehouse in records:
        inventory_by_sku[inventory.product_sku].append(
            (inventory, warehouse)
        )

    # 4. Investigate each requested SKU.
    products = []
    overall_sufficient = True

    for sku, required_quantity in requirements.items():
        sku_records = inventory_by_sku.get(sku, [])

        if not sku_records:
            overall_sufficient = False

            products.append({
                "sku": sku,
                "required_quantity": required_quantity,
                "inventory_found": False,
                "total_available": 0,
                "shortfall": required_quantity,
                "sufficient": False,
                "warehouses": [],
            })

            continue

        warehouse_results = []
        total_available = 0

        for inventory, warehouse in sku_records:
            available = inventory.available_quantity

            # Detect invalid inventory data instead of
            # reporting misleading availability.
            if (
                inventory.quantity_on_hand < 0
                or inventory.quantity_reserved < 0
                or inventory.quantity_reserved > inventory.quantity_on_hand
            ):
                raise ValueError(
                    f"Invalid inventory quantities for SKU {sku} "
                    f"in warehouse {warehouse.warehouse_code}."
                )

            total_available += available

            warehouse_results.append({
                "warehouse_code": warehouse.warehouse_code,
                "warehouse_name": warehouse.name,
                "quantity_on_hand": inventory.quantity_on_hand,
                "quantity_reserved": inventory.quantity_reserved,
                "available_quantity": available,
                "reorder_level": inventory.reorder_level,
                "below_reorder_level": (
                    available < inventory.reorder_level
                ),
            })

        shortfall = max(0, required_quantity - total_available)
        sufficient = shortfall == 0

        if not sufficient:
            overall_sufficient = False

        products.append({
            "sku": sku,
            "product_name": sku_records[0][0].product_name,
            "required_quantity": required_quantity,
            "inventory_found": True,
            "total_available": total_available,
            "shortfall": shortfall,
            "sufficient": sufficient,
            "warehouses": warehouse_results,
        })

    # 5. Return read-only investigation report.
    return {
        "agent": "inventory",
        "success": True,
        "overall_sufficient": overall_sufficient,
        "products_investigated": len(products),
        "products": products,
    }

from app.config import settings


client = genai.Client(api_key=settings.gemini_api_key)

MODEL_CHAINS = {
    "routine": [
        settings.gemini_routine_model,
        settings.gemini_complex_model,
        settings.gemini_fallback_model,
    ],
    "complex": [
        settings.gemini_complex_model,
        settings.gemini_fallback_model,
    ],

def is_temporary_error(exc: Exception) -> bool:
    error = str(exc).upper()
    return any(
        marker in error
        for marker in (
            "503",
            "UNAVAILABLE",
            "429",
            "RESOURCE_EXHAUSTED",
            "500",
            "504",
            "INTERNAL",
        )
    )


def generate_response(prompt: str, task_type: str = "routine") -> dict:
    if task_type not in MODEL_CHAINS:
        raise ValueError(f"Unsupported task type: {task_type}")

    last_error = None

    for model_name in MODEL_CHAINS[task_type]:
        try:
            response = client.models.generate_content(
                model=model_name,
                contents=prompt,
            )

            return {
                "success": True,
                "model": model_name,
                "fallback_used": model_name != MODEL_CHAINS[task_type][0],
                "text": response.text,
            }

        except Exception as exc:
            last_error = exc

            if not is_temporary_error(exc):
                raise

    raise RuntimeError(
        f"All models failed for {task_type} tasks. "
        f"Last error: {last_error}"
    ) from last_error

from collections import defaultdict

from sqlmodel import Session, select

from app.db.models import (
    Supplier, SupplierProduct, PurchaseOrder, PurchaseOrderItem
)

def investigate_procurement(session: Session, required_items: list[dict],) -> dict:
    """
    Read-only procurement investigation.

    Compares active suppliers, prices, availability,
    minimum order quantities, lead times, and existing
    purchase orders.

    Does not create purchase orders or modify inventory.
    """

    if not required_items:
        raise ValueError("Required items cannot be empty.")

    # 1. Validate and combine duplicate SKUs.
    requirements = {}

    for item in required_items:
        sku = item.get("sku")
        quantity = item.get("quantity")

        if not isinstance(sku, str) or not sku.strip():
            raise ValueError("Every item must have a valid SKU.")

        if (
            not isinstance(quantity, int)
            or isinstance(quantity, bool)
            or quantity <= 0
        ):
            raise ValueError(f"Invalid quantity for SKU {sku}.")

        sku = sku.strip()
        requirements[sku] = requirements.get(sku, 0) + quantity

    requested_skus = list(requirements.keys())

    # 2. Retrieve all active supplier offerings in one query.
    supplier_statement = (
        select(SupplierProduct, Supplier)
        .join(
            Supplier,
            SupplierProduct.supplier_id == Supplier.id,
        )
        .where(
            SupplierProduct.product_sku.in_(requested_skus),
            Supplier.is_active.is_(True),
        )
        .order_by(
            SupplierProduct.product_sku,
            SupplierProduct.unit_price,
        )
    )

    supplier_records = session.exec(supplier_statement).all()

    suppliers_by_sku = defaultdict(list)

    for product, supplier in supplier_records:
        suppliers_by_sku[product.product_sku].append(
            (product, supplier)
        )

    # 3. Retrieve existing purchase orders for requested SKUs.
    po_statement = (
        select(PurchaseOrderItem, PurchaseOrder, Supplier)
        .join(
            PurchaseOrder,
            PurchaseOrderItem.purchase_order_id == PurchaseOrder.id,
        )
        .join(
            Supplier,
            PurchaseOrder.supplier_id == Supplier.id,
        )
        .where(
            PurchaseOrderItem.product_sku.in_(requested_skus),
            PurchaseOrder.status != "cancelled",
        )
        .order_by(PurchaseOrder.expected_delivery)
    )

    po_records = session.exec(po_statement).all()

    purchase_orders_by_sku = defaultdict(list)

    for item, purchase_order, supplier in po_records:
        purchase_orders_by_sku[item.product_sku].append({
            "po_number": purchase_order.po_number,
            "supplier_code": supplier.supplier_code,
            "supplier_name": supplier.name,
            "status": purchase_order.status.value,
            "quantity": item.quantity,
            "unit_price": str(item.unit_price),
            "expected_delivery": (
                purchase_order.expected_delivery.isoformat()
                if purchase_order.expected_delivery
                else None
            ),
        })

    # 4. Investigate each requested product.
    products = []

    for sku, required_quantity in requirements.items():
        records = suppliers_by_sku.get(sku, [])

        supplier_options = []

        for product, supplier in records:
            meets_moq = (
                required_quantity >= product.minimum_order_quantity
            )

            suggested_order_quantity = max(
                required_quantity,
                product.minimum_order_quantity,
            )

            supplier_options.append({
                "supplier_code": supplier.supplier_code,
                "supplier_name": supplier.name,
                "unit_price": str(product.unit_price),
                "lead_time_days": product.lead_time_days,
                "minimum_order_quantity": (
                    product.minimum_order_quantity
                ),
                "is_available": product.is_available,
                "meets_moq": meets_moq,
                "suggested_order_quantity": (
                    suggested_order_quantity
                    if product.is_available
                    else None
                ),
                "estimated_total_cost": (
                    str(
                        product.unit_price
                        * suggested_order_quantity
                    )
                    if product.is_available
                    else None
                ),
                "eligible": product.is_available,
            })

        # Sort available options by price, then lead time.
        supplier_options.sort(
            key=lambda option: (
                not option["eligible"],
                float(option["unit_price"]),
                option["lead_time_days"],
            )
        )

        available_options = [
            option for option in supplier_options
            if option["eligible"]
        ]

        products.append({
            "sku": sku,
            "required_quantity": required_quantity,
            "supplier_found": len(records) > 0,
            "available_supplier_count": len(available_options),
            "supplier_options": supplier_options,
            "existing_purchase_orders": (
                purchase_orders_by_sku.get(sku, [])
            ),
            "investigation_status": (
                "options_found"
                if available_options
                else "no_available_supplier"
            ),
        })

    # 5. Return investigation report.
    return {
        "agent": "procurement",
        "success": True,
        "products_investigated": len(products),
        "products": products,
        "read_only": True,
    }

from sqlmodel import Session, select
from sqlalchemy.orm import selectinload

from app.db.models import Order, Inventory, Warehouse


def verify_fulfillment(
    session: Session,
    fulfillment_result: dict,
) -> dict:
    """
    Independently verifies fulfillment recommendations and
    warehouse allocations against current database records.

    Read-only: performs no operational changes.
    """

    if not fulfillment_result:
        raise ValueError("Fulfillment result is required.")

    order_number = fulfillment_result.get("order_number")

    if not order_number:
        raise ValueError("Fulfillment result has no order number.")

    if not fulfillment_result.get("success"):
        return {
            "agent": "verification",
            "success": True,
            "verified": False,
            "issues": ["The fulfillment evaluation was unsuccessful."],
            "read_only": True,
            "execution_performed": False,
        }

    # 1. Independently retrieve the order.
    statement = (
        select(Order)
        .where(Order.order_number == order_number)
        .options(selectinload(Order.items))
    )
    order = session.exec(statement).first()

    if order is None:
        return {
            "agent": "verification",
            "success": True,
            "verified": False,
            "issues": [f"Order {order_number} does not exist."],
            "read_only": True,
            "execution_performed": False,
        }

    issues = []
    checks = []

    # 2. Verify order quantities.
    expected_items = {
        item.product_sku: item.quantity
        for item in order.items
    }

    fulfillment_items = fulfillment_result.get("items", [])

    reported_items = {
        item.get("sku"): item.get("required_quantity")
        for item in fulfillment_items
    }

    if expected_items != reported_items:
        issues.append("Fulfillment quantities do not match the order.")
    else:
        checks.append("Order items and required quantities match.")

    # 3. Independently retrieve active warehouse inventory.
    inventory_statement = (
        select(Inventory)
        .join(
            Warehouse,
            Inventory.warehouse_id == Warehouse.id,
        )
        .where(Warehouse.is_active.is_(True))
    )

    inventory_records = session.exec(inventory_statement).all()

    inventory_by_sku = {}
    warehouse_inventory = {}

    for record in inventory_records:
        available = (
            record.quantity_on_hand - record.quantity_reserved
        )

        if (
            record.quantity_on_hand < 0
            or record.quantity_reserved < 0
            or record.quantity_reserved > record.quantity_on_hand
        ):
            issues.append(
                f"Invalid inventory quantities for {record.product_sku}."
            )
            continue

        warehouse = session.get(Warehouse, record.warehouse_id)

        if warehouse is None or not warehouse.is_active:
            continue

        inventory_by_sku[record.product_sku] = (
            inventory_by_sku.get(record.product_sku, 0)
            + available
        )

        warehouse_inventory[
            (record.product_sku, warehouse.warehouse_code)
        ] = {
            "available_quantity": available,
            "reorder_level": record.reorder_level,
        }

    # 4. Verify aggregate quantities and warehouse allocations.
    for item in fulfillment_items:
        sku = item.get("sku")
        required = item.get("required_quantity")
        reported_available = item.get("available_inventory")
        reported_allocated = item.get("allocated_quantity")
        reported_shortfall = item.get("inventory_shortfall")
        actual_available = inventory_by_sku.get(sku, 0)

        if not isinstance(required, int) or isinstance(required, bool) or required <= 0:
            issues.append(f"Invalid required quantity for {sku}.")
            continue

        if reported_available != actual_available:
            issues.append(
                f"Reported inventory for {sku} does not match database."
            )

        allocations = item.get("warehouse_allocations", [])

        allocated_total = 0
        seen_warehouses = set()

        for allocation in allocations:
            warehouse_code = allocation.get("warehouse_code")
            quantity = allocation.get("allocated_quantity")

            if (
                not isinstance(quantity, int)
                or isinstance(quantity, bool)
                or quantity <= 0
            ):
                issues.append(
                    f"Invalid allocation quantity for {sku} at {warehouse_code}."
                )
                continue

            if warehouse_code in seen_warehouses:
                issues.append(
                    f"Duplicate warehouse allocation for {sku} at {warehouse_code}."
                )
                continue

            seen_warehouses.add(warehouse_code)

            actual_warehouse = warehouse_inventory.get(
                (sku, warehouse_code)
            )

            if actual_warehouse is None:
                issues.append(
                    f"Allocation references inactive or unknown warehouse "
                    f"{warehouse_code} for {sku}."
                )
                continue

            warehouse_available = actual_warehouse["available_quantity"]
            reorder_level = actual_warehouse["reorder_level"]

            if quantity > warehouse_available:
                issues.append(
                    f"Allocation exceeds available stock for {sku} "
                    f"at {warehouse_code}."
                )
                continue

            expected_remaining = warehouse_available - quantity

            if allocation.get("remaining_available") != expected_remaining:
                issues.append(
                    f"Incorrect remaining inventory for {sku} at {warehouse_code}."
                )

            expected_below_reorder = expected_remaining < reorder_level

            if (
                allocation.get("below_reorder_level_after_allocation")
                != expected_below_reorder
            ):
                issues.append(
                    f"Incorrect reorder-level assessment for {sku} "
                    f"at {warehouse_code}."
                )

            allocated_total += quantity

        if reported_allocated != allocated_total:
            issues.append(
                f"Allocated quantity does not match warehouse allocations for {sku}."
            )

        actual_shortfall = max(0, required - allocated_total)

        if reported_shortfall != actual_shortfall:
            issues.append(
                f"Reported shortfall is incorrect for {sku}."
            )

        if allocated_total > required:
            issues.append(
                f"Allocated quantity exceeds the requirement for {sku}."
            )

        action = item.get("recommended_action")

        if actual_shortfall == 0:
            if action != "fulfill_from_allocated_inventory":
                issues.append(
                    f"Incorrect fulfillment action for {sku}."
                )
        elif action not in (
            "procure_shortfall",
            "requires_manual_review",
        ):
            issues.append(
                f"Invalid fulfillment action for {sku}."
            )

    if not any(
        "inventory" in issue.lower()
        or "allocation" in issue.lower()
        or "warehouse" in issue.lower()
        or "reorder" in issue.lower()
        or "shortfall" in issue.lower()
        for issue in issues
    ):
        checks.append(
            "Warehouse allocations match available stock and reorder levels."
        )

    # 5. Verify overall fulfillment status.
    actual_all_sufficient = all(
        item.get("required_quantity", 0)
        == item.get("allocated_quantity", 0)
        for item in fulfillment_items
    )

    all_items_have_path = all(
        item.get("recommended_action") != "requires_manual_review"
        for item in fulfillment_items
    )

    if actual_all_sufficient:
        expected_status = "inventory_fulfillment_possible"
    elif all_items_have_path:
        expected_status = "procurement_required"
    else:
        expected_status = "manual_review_required"

    if fulfillment_result.get("overall_status") != expected_status:
        issues.append("Overall fulfillment status is inconsistent.")

    # 6. Confirm that no operational execution was performed.
    if fulfillment_result.get("execution_performed") is not False:
        issues.append("Fulfillment result does not confirm read-only execution.")

    if fulfillment_result.get("read_only") is not True:
        issues.append("Fulfillment result does not confirm read-only operation.")

    verified = len(issues) == 0

    if verified:
        checks.append("Fulfillment actions and overall status are consistent.")

    return {
        "agent": "verification",
        "success": True,
        "verified": verified,
        "order_number": order_number,
        "checks_passed": checks,
        "issues": issues,
        "overall_status": (
            "verification_passed"
            if verified
            else "verification_failed"
        ),
        "read_only": True,
        "execution_performed": False,
    }

router = APIRouter(
    prefix="/api/approvals",
    tags=["Approvals"],

# Development-only identity. Replace with authenticated identity
# resolution before exposing this API outside localhost.
DEV_APPROVER_EMAIL = "approver@enterpriseops.local"


@router.get("", response_model=list[ApprovalResponse])
def list_pending_approvals(
    session: Session = Depends(get_session),
):
    """List pending approval requests."""

    return session.exec(
        select(ApprovalRequest)
        .where(ApprovalRequest.status == ApprovalStatus.PENDING)
        .order_by(ApprovalRequest.requested_at.desc())
    ).all()


@router.get("/{approval_id}", response_model=ApprovalResponse)
def get_approval(
    approval_id: uuid.UUID,
    session: Session = Depends(get_session),
):
    """Retrieve an approval request."""

    approval = session.get(ApprovalRequest, approval_id)

    if not approval:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Approval request not found.",
        )

    return approval


@router.post(
    "/{approval_id}/decision",
    response_model=ApprovalResponse,
)
def decide_approval(
    approval_id: uuid.UUID,
    decision_data: ApprovalDecision,
    session: Session = Depends(get_session),
):
    """Approve or reject a pending request."""

    reviewer = session.exec(
        select(User).where(
            User.email == DEV_APPROVER_EMAIL,
            User.is_active.is_(True),
        )
    ).first()

    if not reviewer:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Development approver account not found or inactive.",
        )

    try:
        return review_approval(
            session=session,
            approval_id=approval_id,
            reviewer_id=reviewer.id,
            decision_data=decision_data,
        )

    except ApprovalNotFoundError as exc:
        raise HTTPException(status_code=404, detail=str(exc))

    except UnauthorizedApproverError as exc:
        raise HTTPException(status_code=403, detail=str(exc))

    except InvalidApprovalDecisionError as exc:
        raise HTTPException(status_code=400, detail=str(exc))

    except ApprovalAlreadyReviewedError as exc:
        raise HTTPException(status_code=409, detail=str(exc))

from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, model_validator


SpecialistAgent = Literal[
    "order",
    "inventory",
    "procurement",
    "fulfillment",
    "verification",
]


class ExecutionStep(BaseModel):
    model_config = ConfigDict(extra="forbid")

    step: int = Field(ge=1)
    agent: SpecialistAgent
    action: str = Field(min_length=1, max_length=500)
    expected_output: str = Field(min_length=1, max_length=500)
    depends_on: list[int] = Field(default_factory=list)


class ExecutionPlan(BaseModel):
    model_config = ConfigDict(extra="forbid")

    task_type: Literal["routine", "complex"]
    problem_summary: str = Field(min_length=1, max_length=1000)
    priority: Literal["low", "medium", "high", "critical"]
    agents_required: list[SpecialistAgent] = Field(min_length=1)
    execution_plan: list[ExecutionStep] = Field(min_length=1)
    requires_human_approval: bool
    verification_required: bool

    @model_validator(mode="after")
    def validate_plan(self):
        steps = self.execution_plan
        step_numbers = [step.step for step in steps]

        if len(step_numbers) != len(set(step_numbers)):
            raise ValueError("Execution step numbers must be unique.")

        if step_numbers != sorted(step_numbers):
            raise ValueError("Execution steps must be in ascending order.")

        available_steps = set()

        for step in steps:
            if step.step in step.depends_on:
                raise ValueError(
                    f"Step {step.step} cannot depend on itself."
                )

            if len(step.depends_on) != len(set(step.depends_on)):
                raise ValueError(
                    f"Step {step.step} has duplicate dependencies."
                )

            for dependency in step.depends_on:
                if dependency not in available_steps:
                    raise ValueError(
                        f"Step {step.step} depends on invalid or "
                        f"unfinished step {dependency}."
                    )

            available_steps.add(step.step)

        step_agents = {step.agent for step in steps}
        required_agents = set(self.agents_required)

        if len(required_agents) != len(self.agents_required):
            raise ValueError("agents_required contains duplicates.")

        if step_agents != required_agents:
            raise ValueError(
                "agents_required must match the agents in execution_plan."
            )

        if self.verification_required:
            if "verification" not in required_agents:
                raise ValueError(
                    "Verification is required but no verification "
                    "step was included."
                )

        return self

def persist_agent_execution(

    *,
    task_id: UUID,

) -> AgentRun:

        metadata_json={
            "agent": agent,
            "status": run.status.value,
        },
    )
    session.add(audit)
    session.flush()

    return run

import uuid

    User,

# Only these roles may review approval requests.
APPROVER_ROLES = {"admin", "manager", "approver"}


class ApprovalNotFoundError(Exception):
    pass


class UnauthorizedApproverError(Exception):
    pass


class InvalidApprovalDecisionError(Exception):
    pass


class ApprovalAlreadyReviewedError(Exception):
    pass


def utc_now():
    return datetime.now(timezone.utc)


def review_approval(
    session: Session,
    approval_id: uuid.UUID,
    reviewer_id: uuid.UUID,
    decision_data: ApprovalDecision,
) -> ApprovalRequest:
    """Review a pending approval and record an audit event."""

    if decision_data.decision not in {
        ApprovalStatus.APPROVED,
        ApprovalStatus.REJECTED,
    }:
        raise InvalidApprovalDecisionError(
            "Only approved or rejected decisions are permitted."
        )

    # Verify the reviewer is active and has an authorized role.
    reviewer = session.exec(
        select(User)
        .where(
            User.id == reviewer_id,
            User.is_active.is_(True),
        )
        .with_for_update()
    ).first()

    if not reviewer or reviewer.role not in APPROVER_ROLES:
        raise UnauthorizedApproverError(
            "The user is not authorized to review approval requests."
        )

    # Lock the approval row to prevent concurrent reviews.
    approval = session.exec(
        select(ApprovalRequest)
        .where(ApprovalRequest.id == approval_id)
        .with_for_update()
    ).first()

    if not approval:
        raise ApprovalNotFoundError("Approval request not found.")

    if approval.status != ApprovalStatus.PENDING:
        raise ApprovalAlreadyReviewedError(
            f"Approval request is already {approval.status.value}."
        )

    reviewed_at = utc_now()
    previous_status = approval.status

    approval.status = decision_data.decision
    approval.reviewed_at = reviewed_at
    approval.reviewed_by = reviewer.id
    approval.reviewer_comment = decision_data.reviewer_comment

    audit_log = AuditLog(
        task_id=approval.task_id,
        actor_type=ActorType.USER,
        actor_id=reviewer.id,
        action=f"approval_{decision_data.decision.value}",
        entity_type="approval_request",
        entity_id=approval.id,
        description=(
            f"Approval request {approval.id} was "
            f"{decision_data.decision.value} by {reviewer.email}."
        ),
        metadata_json={
            "previous_status": previous_status.value,
            "new_status": decision_data.decision.value,
            "action_type": approval.action_type,
            "reviewer_email": reviewer.email,
            "reviewer_comment": decision_data.reviewer_comment,
            "execution_performed": False,
        },
    )

    session.add(approval)
    session.add(audit_log)

    try:
        # Both the decision and audit record commit together.
        session.commit()
        session.refresh(approval)
        return approval

    except Exception:
        session.rollback()
        raise

import re
import uuid
from datetime import datetime, timezone

from sqlalchemy.engine import make_url
from sqlmodel import Session, select
from langgraph.checkpoint.postgres import PostgresSaver

from app.config import settings
from app.db.models import OperationalTask, TaskStatus, User
from app.schemas.task import TaskCreate
from app.agents.graph import build_graph


def utc_now():
    return datetime.now(timezone.utc)


def extract_order_number(objective: str, input_context: dict) -> str | None:
    """Find an order number in the supplied context or objective."""
    order_number = input_context.get("order_number")

    if isinstance(order_number, str) and order_number.strip():
        return order_number.strip()

    match = re.search(r"\bORD-\d+\b", objective, re.IGNORECASE)
    return match.group(0).upper() if match else None


def execute_task_workflow(task: OperationalTask) -> dict:
    """Run the existing read-only LangGraph workflow with PostgreSQL persistence."""

    database_url = make_url(settings.database_url)
    checkpoint_url = database_url.set(
        drivername="postgresql"
    ).render_as_string(hide_password=False)

    thread_id = str(task.id)

    initial_state = {
        "task_id": task.id,
        "objective": task.objective,
        "order_number": task.input_context["order_number"],
        "task_context": task.input_context,
        "task_status": "running",
        "messages": [],
        "agent_results": [],
        "errors": [],
        "audit_events": [],
    }

    config = {
        "configurable": {
            "thread_id": thread_id,
        }
    }

    with PostgresSaver.from_conn_string(checkpoint_url) as checkpointer:
        checkpointer.setup()

        graph = build_graph(checkpointer=checkpointer)
        result = graph.invoke(initial_state, config=config)

    return result["final_report"]


def create_and_execute_task(
    session: Session,
    user_id: uuid.UUID,
    task_data: TaskCreate,
) -> OperationalTask:
    """Create a task, execute its workflow, and persist the result."""

    user = session.exec(
        select(User).where(
            User.id == user_id,
            User.is_active.is_(True),
        )
    ).first()

    if not user:
        raise ValueError("Active user not found.")

    input_context = dict(task_data.input_context)

    order_number = extract_order_number(
        task_data.objective,
        input_context,
    )

    if not order_number:
        raise ValueError(
            "An order number (for example, ORD-1042) is required "
            "for the current workflow."
        )

    input_context["order_number"] = order_number

    task = OperationalTask(
        task_key=f"TASK-{uuid.uuid4().hex[:12].upper()}",
        user_id=user.id,
        objective=task_data.objective,
        priority=task_data.priority,
        input_context=input_context,
        status=TaskStatus.RUNNING,
    )

    session.add(task)
    session.commit()
    session.refresh(task)

    try:
        report = execute_task_workflow(task)

        task.final_report = report
        workflow_status = report.get("task_status")

        if workflow_status == "completed":
            task.status = TaskStatus.COMPLETED

        elif workflow_status == "waiting_for_approval":
            task.status = TaskStatus.WAITING_APPROVAL

        else:
            task.status = TaskStatus.FAILED

        if task.status in {
            TaskStatus.COMPLETED,
            TaskStatus.FAILED,
        }:
            task.completed_at = utc_now()
        else:
            task.completed_at = None

    except Exception as exc:
        task.status = TaskStatus.FAILED
        task.final_report = {
            "task_id": str(task.id),
            "order_number": order_number,
            "task_status": "failed",
            "error": str(exc),
            "read_only": True,
            "execution_performed": False,
        }
        task.completed_at = utc_now()

    task.updated_at = utc_now()

    session.add(task)
    session.commit()
    session.refresh(task)

    return task

        "warehouse_code": "WH-BLR-01",
        "name": "Bangalore Distribution Center",

STOCK = {
    "WH-CHN-01": [
        ("SKU-1001", "Industrial Motor", 2, 0, 5),
        ("SKU-1002", "Control Unit", 10, 2, 5),
        ("SKU-1003", "Power Adapter", 50, 5, 10),
        ("SKU-1004", "Sensor Module", 20, 2, 5),
    ],
    "WH-BLR-01": [
        ("SKU-1001", "Industrial Motor", 8, 1, 5),
        ("SKU-1002", "Control Unit", 1, 0, 5),
        ("SKU-1003", "Power Adapter", 30, 3, 10),
        ("SKU-1004", "Sensor Module", 12, 2, 5),
    ],
}


def seed_inventory():
    with Session(engine) as session:

        warehouses = {}

        for data in WAREHOUSES:
            warehouse = session.exec(
                select(Warehouse).where(
                    Warehouse.warehouse_code == data["warehouse_code"]
                )
            ).first()

            if warehouse is None:
                warehouse = Warehouse(**data)
                session.add(warehouse)
                session.flush()
                print(f"Created warehouse: {warehouse.warehouse_code}")
            else:
                print(f"Using existing warehouse: {warehouse.warehouse_code}")

            warehouses[data["warehouse_code"]] = warehouse

        for warehouse_code, products in STOCK.items():
            warehouse = warehouses[warehouse_code]

            for sku, name, on_hand, reserved, reorder in products:
                existing = session.exec(
                    select(Inventory).where(
                        Inventory.warehouse_id == warehouse.id,
                        Inventory.product_sku == sku,
                    )
                ).first()

                if existing:
                    print(f"Skipping existing stock: {warehouse_code} / {sku}")
                    continue

                session.add(
                    Inventory(
                        warehouse_id=warehouse.id,
                        product_sku=sku,
                        product_name=name,
                        quantity_on_hand=on_hand,
                        quantity_reserved=reserved,
                        reorder_level=reorder,
                        last_updated=datetime.now(timezone.utc),
                    )
                )

                print(f"Added stock: {warehouse_code} / {sku}")

        session.commit()
        print("\nInventory seeding completed.")

import json

from sqlmodel import Session

from app.db.session import engine
from app.agents.fulfillment_agent import evaluate_fulfillment
from app.agents.verification_agent import verify_fulfillment


def main():
    with Session(engine) as session:

        # First generate a fulfillment recommendation.
        fulfillment_result = evaluate_fulfillment(
            session,
            "ORD-1042",
        )

        # Independently verify that recommendation.
        verification_result = verify_fulfillment(
            session,
            fulfillment_result,
        )

        print("\nVERIFICATION AGENT RESULT\n")
        print(json.dumps(
            verification_result,
            indent=4,
            default=str,
        ))

        assert verification_result["success"] is True
        assert verification_result["verified"] is True
        assert verification_result["read_only"] is True

        print("\nVERIFICATION AGENT TEST PASSED")


if __name__ == "__main__":
    main()
